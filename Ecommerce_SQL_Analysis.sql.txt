-- E-COMMERCE SALES ANALYSIS  (one database, everything inside it)
-- Tested on SQLite. Works on MySQL/PostgreSQL with small notes below.
--   MySQL:      CREATE DATABASE ecommerce_db; USE ecommerce_db;
--   PostgreSQL: CREATE DATABASE ecommerce_db;  then  \c ecommerce_db
--   SQLite:     skip the two lines above (the database is just a file)

-- ---------------------------------------------------------------------
-- STEP 1: SCHEMA
-- ---------------------------------------------------------------------
DROP TABLE IF EXISTS order_items;
DROP TABLE IF EXISTS orders;
DROP TABLE IF EXISTS products;
DROP TABLE IF EXISTS customers;

CREATE TABLE customers (
  customer_id INT PRIMARY KEY,
  name        VARCHAR(100) NOT NULL,
  city        VARCHAR(50),
  signup_date DATE
);

CREATE TABLE products (
  product_id INT PRIMARY KEY,
  name       VARCHAR(100) NOT NULL,
  category   VARCHAR(50),
  price      DECIMAL(10,2) CHECK (price > 0)
);

CREATE TABLE orders (
  order_id    INT PRIMARY KEY,
  customer_id INT NOT NULL,
  order_date  DATE NOT NULL,
  status      VARCHAR(20),
  FOREIGN KEY (customer_id) REFERENCES customers(customer_id)
);

CREATE TABLE order_items (
  order_id   INT,
  product_id INT,
  quantity   INT CHECK (quantity > 0),
  price      DECIMAL(10,2),          -- price at time of sale
  PRIMARY KEY (order_id, product_id),
  FOREIGN KEY (order_id)   REFERENCES orders(order_id),
  FOREIGN KEY (product_id) REFERENCES products(product_id)
);

-- ---------------------------------------------------------------------
-- STEP 3: DATA CLEANING CHECKS
-- ---------------------------------------------------------------------
-- Duplicate customers by name
SELECT name, COUNT(*) AS cnt FROM customers GROUP BY name HAVING COUNT(*) > 1;

-- NULL checks
SELECT COUNT(*) AS missing_city FROM customers WHERE city IS NULL;

-- Status labels with CASE
SELECT order_id,
       CASE status
         WHEN 'Delivered' THEN 'Completed'
         WHEN 'Shipped'   THEN 'In transit'
         ELSE 'Not counted'
       END AS status_group
FROM orders;

-- ---------------------------------------------------------------------
-- STEP 4: BASIC QUESTIONS
-- (revenue counts only non-cancelled orders)
-- ---------------------------------------------------------------------

-- Q1. Total revenue
SELECT SUM(oi.quantity * oi.price) AS total_revenue
FROM orders o
JOIN order_items oi ON o.order_id = oi.order_id
WHERE o.status <> 'Cancelled';

-- Q2. Revenue per month
-- SQLite: SUBSTR(order_date,1,7)   MySQL: DATE_FORMAT(order_date,'%Y-%m')
-- PostgreSQL: TO_CHAR(order_date,'YYYY-MM')
SELECT SUBSTR(o.order_date, 1, 7) AS month,
       SUM(oi.quantity * oi.price) AS revenue
FROM orders o
JOIN order_items oi ON o.order_id = oi.order_id
WHERE o.status <> 'Cancelled'
GROUP BY SUBSTR(o.order_date, 1, 7)
ORDER BY month;

-- Q3. Top 10 best-selling products (by units)
SELECT p.name, SUM(oi.quantity) AS units_sold
FROM order_items oi
JOIN products p ON p.product_id = oi.product_id
JOIN orders o   ON o.order_id = oi.order_id
WHERE o.status <> 'Cancelled'
GROUP BY p.product_id, p.name
ORDER BY units_sold DESC
LIMIT 10;

-- Q4. Customers per city
SELECT city, COUNT(*) AS customers
FROM customers
GROUP BY city
ORDER BY customers DESC;

-- ---------------------------------------------------------------------
-- STEP 5: INTERMEDIATE QUESTIONS
-- ---------------------------------------------------------------------

-- Q5. Revenue by product category
SELECT p.category, SUM(oi.quantity * oi.price) AS revenue
FROM order_items oi
JOIN products p ON p.product_id = oi.product_id
JOIN orders o   ON o.order_id = oi.order_id
WHERE o.status <> 'Cancelled'
GROUP BY p.category
ORDER BY revenue DESC;

-- Q6. Customers with more than 3 orders (use > 5 on a bigger dataset)
SELECT c.name, COUNT(o.order_id) AS orders_count
FROM customers c
JOIN orders o ON o.customer_id = c.customer_id
GROUP BY c.customer_id, c.name
HAVING COUNT(o.order_id) > 3;

-- Q7. Products that were never sold
SELECT p.product_id, p.name
FROM products p
LEFT JOIN order_items oi ON oi.product_id = p.product_id
WHERE oi.product_id IS NULL;

-- Q8. Customers who never placed an order
SELECT c.name
FROM customers c
LEFT JOIN orders o ON o.customer_id = c.customer_id
WHERE o.order_id IS NULL;

-- ---------------------------------------------------------------------
-- STEP 6: ADVANCED QUESTIONS (CTEs + window functions)
-- Needs SQLite 3.25+, MySQL 8+, or PostgreSQL
-- ---------------------------------------------------------------------

-- Q9. Rank customers by total spending
WITH spending AS (
  SELECT c.customer_id, c.name,
         SUM(oi.quantity * oi.price) AS total_spent
  FROM customers c
  JOIN orders o       ON o.customer_id = c.customer_id
  JOIN order_items oi ON oi.order_id = o.order_id
  WHERE o.status <> 'Cancelled'
  GROUP BY c.customer_id, c.name
)
SELECT name, total_spent,
       RANK()       OVER (ORDER BY total_spent DESC) AS rnk,
       DENSE_RANK() OVER (ORDER BY total_spent DESC) AS dense_rnk
FROM spending;

-- Q10. Month-over-month revenue growth (LAG)
WITH monthly AS (
  SELECT SUBSTR(o.order_date, 1, 7) AS month,
         SUM(oi.quantity * oi.price) AS revenue
  FROM orders o
  JOIN order_items oi ON o.order_id = oi.order_id
  WHERE o.status <> 'Cancelled'
  GROUP BY SUBSTR(o.order_date, 1, 7)
)
SELECT month, revenue,
       LAG(revenue) OVER (ORDER BY month) AS prev_month,
       ROUND(100.0 * (revenue - LAG(revenue) OVER (ORDER BY month))
             / LAG(revenue) OVER (ORDER BY month), 1) AS growth_pct
FROM monthly;

-- Q11. Running total of sales
WITH monthly AS (
  SELECT SUBSTR(o.order_date, 1, 7) AS month,
         SUM(oi.quantity * oi.price) AS revenue
  FROM orders o
  JOIN order_items oi ON o.order_id = oi.order_id
  WHERE o.status <> 'Cancelled'
  GROUP BY SUBSTR(o.order_date, 1, 7)
)
SELECT month, revenue,
       SUM(revenue) OVER (ORDER BY month) AS running_total
FROM monthly;

-- Q12. Average order value per customer
WITH order_totals AS (
  SELECT o.order_id, o.customer_id,
         SUM(oi.quantity * oi.price) AS order_total
  FROM orders o
  JOIN order_items oi ON oi.order_id = o.order_id
  WHERE o.status <> 'Cancelled'
  GROUP BY o.order_id, o.customer_id
)
SELECT c.name, ROUND(AVG(ot.order_total), 2) AS avg_order_value
FROM order_totals ot
JOIN customers c ON c.customer_id = ot.customer_id
GROUP BY c.customer_id, c.name
ORDER BY avg_order_value DESC;

-- Q13. Customer segments by spend (CASE) + average order value per segment
WITH order_totals AS (
  SELECT o.order_id, o.customer_id,
         SUM(oi.quantity * oi.price) AS order_total
  FROM orders o
  JOIN order_items oi ON oi.order_id = o.order_id
  WHERE o.status <> 'Cancelled'
  GROUP BY o.order_id, o.customer_id
),
customer_spend AS (
  SELECT customer_id, SUM(order_total) AS total_spent, AVG(order_total) AS aov
  FROM order_totals
  GROUP BY customer_id
)
SELECT CASE WHEN total_spent >= 10000 THEN 'High'
            WHEN total_spent >= 4000  THEN 'Medium'
            ELSE 'Low' END AS segment,
       COUNT(*) AS customers,
       ROUND(AVG(aov), 2) AS avg_order_value
FROM customer_spend
GROUP BY 1;

-- Q14. Repeat purchase: customers who bought again within 30 days
-- SQLite: JULIANDAY(a) - JULIANDAY(b)   MySQL: DATEDIFF(a, b)
-- PostgreSQL: (a - b)
WITH ordered AS (
  SELECT customer_id, order_date,
         LAG(order_date) OVER (PARTITION BY customer_id ORDER BY order_date) AS prev_date
  FROM orders
  WHERE status <> 'Cancelled'
)
SELECT DISTINCT c.name
FROM ordered x
JOIN customers c ON c.customer_id = x.customer_id
WHERE x.prev_date IS NOT NULL
    AND DATEDIFF(x.order_date, x.prev_date) <= 30;

-- Reusable report views
CREATE VIEW v_monthly_revenue AS
SELECT SUBSTR(o.order_date, 1, 7) AS month,
       SUM(oi.quantity * oi.price) AS revenue
FROM orders o
JOIN order_items oi ON o.order_id = oi.order_id
WHERE o.status <> 'Cancelled'
GROUP BY SUBSTR(o.order_date, 1, 7);

CREATE VIEW v_customer_spending AS
SELECT c.customer_id, c.name, c.city,
       SUM(oi.quantity * oi.price) AS total_spent
FROM customers c
JOIN orders o       ON o.customer_id = c.customer_id
JOIN order_items oi ON oi.order_id = o.order_id
WHERE o.status <> 'Cancelled'
GROUP BY c.customer_id, c.name, c.city;

SELECT * FROM v_monthly_revenue;
SELECT * FROM v_customer_spending ORDER BY total_spent DESC;
